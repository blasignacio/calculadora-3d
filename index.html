// Reemplaza esta constante con la URL de tu Web App const WEB_APP_URL = "TU_URL_DE_WEB_APP_AQUI"; function guardarEnSheet() { // Captura de valores desde la interfaz const proyecto = document.getElementById("nombrePieza") ? document.getElementById("nombrePieza").value : "Sin nombre"; const tipo = document.getElementById("tipoMaterial") ? document.getElementById("tipoMaterial").value : "PLA"; const peso = document.getElementById("gramosMaterial") ? document.getElementById("gramosMaterial").value : 0; const nombreExtra = document.getElementById("nombreExtra") ? document.getElementById("nombreExtra").value : ""; const costoExtra = document.getElementById("costoExtra") ? document.getElementById("costoExtra").value : 0; const horas = document.getElementById("horas") ? document.getElementById("horas").value : 0; const minutos = document.getElementById("minutos") ? document.getElementById("minutos").value : 0; const cantidad = document.getElementById("cantidad") ? document.getElementById("cantidad").value : 1; const costoUnitario = document.getElementById("costoUnitario") ? document.getElementById("costoUnitario").value : 0; const costoReal = document.getElementById("costoTotal") ? document.getElementById("costoTotal").value : 0; const cotizacionReal = document.getElementById("precioVenta") ? document.getElementById("precioVenta").value : 0; const beneficio = document.getElementById("beneficio") ? document.getElementById("beneficio").value : (cotizacionReal - costoReal); const precioKg = document.getElementById("precioKg") ? document.getElementById("precioKg").value : 0; // Formato para MATERIAL_RESUMEN: Solo el material principal y los gramos const materialResumenFormat = tipo + " (" + peso + "g)"; // Formato para MATERIAL_EXTRA: Insumos adicionales por separado const materialExtraFormat = (nombreExtra && nombreExtra.trim() !== "") ? nombreExtra + " ($ " + costoExtra + ")" : "-"; // Formato para TIEMPO_RESUMEN const tiempoResumenFormat = horas + "h " + minutos + "m"; // Objeto estructurado según las claves exactas de tus encabezados const payload = { PROYECTO: proyecto, MATERIAL_RESUMEN: materialResumenFormat, MATERIAL_EXTRA: materialExtraFormat, TIEMPO_RESUMEN: tiempoResumenFormat, CANTIDAD: cantidad, COSTO_UNITARIO: costoUnitario, COSTO_REAL: costoReal, COTIZACION_REAL: cotizacionReal, BENEFICIO: beneficio, PRECIO_KG: precioKg, PESO: peso, HORAS: horas, MINUTOS: minutos, TIPO: tipo }; fetch(WEB_APP_URL, { method: "POST", mode: "no-cors", headers: { "Content-Type": "application/json" }, body: JSON.stringify(payload) }) .then(() => { alert("Cotización guardada exitosamente en Google Sheets"); cargarHistorial(); }) .catch(error => { console.error("Error al guardar:", error); alert("Hubo un error al guardar la cotización."); }); } function cargarHistorial() { fetch(WEB_APP_URL) .then(response => response.json()) .then(res => { if (res.status === "success") { console.log("Historial recuperado correctamente:", res.data); } }) .catch(err => console.error("Error al obtener historial:", err)); } function eliminarRegistro(nombreProyecto) { const payload = { action: "delete", PROYECTO: nombreProyecto }; fetch(WEB_APP_URL, { method: "POST", mode: "no-cors", headers: { "Content-Type": "application/json" }, body: JSON.stringify(payload) }) .then(() => { alert("Registro eliminado"); cargarHistorial(); }) .catch(err => console.error("Error al eliminar:", err)); }
